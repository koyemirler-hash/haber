rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // ── Yardımcı fonksiyonlar ──────────────────────────────

    function yetkili() {
      return request.auth != null && (
        request.auth.token.email == "koyemirler@gmail.com" ||
        get(/databases/$(database)/documents/users/$(request.auth.uid)).data.rol
          in ["admin", "muhtar", "yardimci"]
      );
    }

    function girisYapti() {
      return request.auth != null;
    }

    // Özel mesaj konuşma ID'si: "uid1_uid2" (sıralı)
    function konusmaKatilimcisi(konusmaId) {
      let parts = konusmaId.split('_');
      return parts.size() == 2 &&
             (parts[0] == request.auth.uid || parts[1] == request.auth.uid);
    }

    // Metin alanı boyut kontrolü
    function textOk(field, maxLen) {
      return field is string && field.size() <= maxLen;
    }

    // ── Kullanıcılar ──────────────────────────────────────
    match /users/{userId} {
      allow read: if true;
      // Sadece kendi profilini oluşturabilir
      allow create: if girisYapti() && request.auth.uid == userId
        && textOk(request.resource.data.name, 60)
        && textOk(request.resource.data.email, 120)
        && textOk(request.resource.data.phone, 20);
      // Güncelleme: kendi profili veya yetkili
      allow update: if girisYapti() &&
        (request.auth.uid == userId || yetkili()) &&
        // rol alanını sadece yetkili değiştirebilir
        (!request.resource.data.diff(resource.data).affectedKeys().hasAny(["rol","blocked"])
          || yetkili());
      allow delete: if yetkili();
    }

    // ── Duyurular ─────────────────────────────────────────
    match /announcements/{id} {
      allow read: if true;
      allow create: if yetkili();
      allow update: if girisYapti();
      allow delete: if yetkili();
      match /comments/{cid} {
        allow read: if true;
        allow create: if girisYapti()
          && textOk(request.resource.data.text, 500);
        allow delete: if girisYapti() &&
          (resource.data.uid == request.auth.uid || yetkili());
        allow update: if false;
      }
    }

    // ── Köy Sohbeti ───────────────────────────────────────
    match /chat/{id} {
      allow read: if true;
      allow create: if girisYapti()
        && textOk(request.resource.data.text, 1000)
        && request.resource.data.uid == request.auth.uid;
      allow delete: if girisYapti() &&
        (resource.data.uid == request.auth.uid || yetkili());
      allow update: if false;
    }

    // ── Özel Mesajlar ─────────────────────────────────────
    match /ozelMesajlar/{konusmaId} {
      allow read, write: if girisYapti() && konusmaKatilimcisi(konusmaId);
      match /mesajlar/{msgId} {
        allow read: if girisYapti() && konusmaKatilimcisi(konusmaId);
        allow create: if girisYapti()
          && konusmaKatilimcisi(konusmaId)
          && request.resource.data.gonderen == request.auth.uid
          && textOk(request.resource.data.text, 1000);
        allow delete: if girisYapti() &&
          (resource.data.gonderen == request.auth.uid || yetkili());
        allow update: if false;
      }
    }

    // ── Nostalji ──────────────────────────────────────────
    match /nostalgia/{id} {
      allow read: if true;
      allow create: if girisYapti()
        && textOk(request.resource.data.title, 200)
        && textOk(request.resource.data.description, 2000);
      allow update: if girisYapti() && (
        yetkili() ||
        request.resource.data.diff(resource.data)
          .affectedKeys().hasOnly(["reactions","commentCount","status","approvedBy","approvedAt"])
      );
      allow delete: if yetkili();
      match /comments/{cid} {
        allow read: if true;
        allow create: if girisYapti()
          && textOk(request.resource.data.text, 500);
        allow delete: if girisYapti() &&
          (resource.data.uid == request.auth.uid || yetkili());
        allow update: if false;
      }
    }

    // ── İlanlar ───────────────────────────────────────────
    match /ilanlar/{id} {
      allow read: if true;
      allow create: if girisYapti()
        && textOk(request.resource.data.title, 150)
        && textOk(request.resource.data.description, 2000)
        && request.resource.data.uid == request.auth.uid;
      allow update: if yetkili();
      allow delete: if girisYapti() &&
        (resource.data.uid == request.auth.uid || yetkili());
    }

    // ── Firmalar ──────────────────────────────────────────
    match /businesses/{id} {
      allow read: if true;
      allow write: if yetkili();
    }

    // ── Rehber ────────────────────────────────────────────
    match /rehber/{userId} {
      allow read: if true;
      allow write: if girisYapti() && request.auth.uid == userId;
    }

    // ── Görüş & Öneri ─────────────────────────────────────
    match /questions/{id} {
      allow read: if yetkili();
      allow create: if girisYapti()
        && textOk(request.resource.data.text, 1000);
      allow update, delete: if yetkili();
    }

    // ── Ayarlar ───────────────────────────────────────────
    match /settings/{id} {
      allow read: if true;
      allow create, delete: if yetkili();
      allow update: if yetkili() ||
        (girisYapti() && id == "anket" &&
         request.resource.data.diff(resource.data).affectedKeys().hasOnly(["oylar"]));
    }

    // ── Şifre Sıfırlama ───────────────────────────────────
    match /resetTalepleri/{id} {
      allow read:   if girisYapti();
      allow create: if girisYapti();
      allow update, delete: if yetkili();
    }

    // ── Hikayeler ─────────────────────────────────────────
    match /stories/{id} {
      allow read: if true;
      allow create: if girisYapti()
        && request.resource.data.uid == request.auth.uid;
      allow update: if girisYapti() && (
        request.resource.data.diff(resource.data).affectedKeys().hasOnly(["viewers"]) ||
        resource.data.uid == request.auth.uid ||
        yetkili()
      );
      allow delete: if girisYapti() &&
        (resource.data.uid == request.auth.uid || yetkili());
    }

    // ── Dernek ────────────────────────────────────────────
    match /dernek/{doc} {
      allow read: if true;
      allow write: if yetkili();
    }

    match /dernekYonetim/{doc} {
      allow read: if true;
      allow write: if yetkili();
    }

  }
}
